<type_var> ::= entier
	    | flottant
	    | texte


<type_comp> ::=  | temperature
	    	         | led
	         

<type_transmition> ::= wifi
	                    | bluetooth

<type-qualifier> ::= constante

<declaration_var> ::= {<type-qualifier>}? <type_var> <identifiant> = <expression>;
<declaration_comp> ::= <type_comp> <identifiant> : <expression>;
<declaration_transmition> ::= <type_transmition> <identifiant> : <expression> {expression}?

<iteration> ::= pour <identifiant>:<expression>:<expression> {
    <expression>
  }
